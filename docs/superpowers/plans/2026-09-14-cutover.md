# cutover Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A public Python package `cutover` that proves two tables are identical (schema, keys, row set, per-column values with float tolerance, per-partition fingerprints) and reports PASS/FAIL as JSON, text and a self-contained HTML file, from a CLI and a Python API.

**Architecture:** Everything becomes a `pyarrow.Table` (`sources.py`); the comparison runs as SQL in an in-process DuckDB connection over the two registered tables (`compare.py`), producing a `Report` dataclass (`report.py`) that renders itself (`html.py`); a thin argparse CLI (`cli.py`) maps the report to exit codes.

**Tech Stack:** Python ≥ 3.10, `duckdb`, `pyarrow`; dev: `uv`, `ruff`, `pytest`; CI: GitHub Actions; release: PyPI trusted publishing.

**Spec:** `docs/superpowers/specs/2026-09-14-cutover-design.md` (in the bsandova.com repo)

## Global Constraints

- Package name `cutover`, licence Apache-2.0, repo `~/Documents/Coding/cutover`, GitHub `sandovabarbora/cutover`.
- Runtime dependencies: `duckdb>=1.0`, `pyarrow>=15`. Nothing else. Extra `bigquery`: `google-cloud-bigquery`.
- Python 3.10–3.13.
- Exit codes: 0 PASS, 1 FAIL, 2 error.
- HTML report: one file, no external resources (test asserts no `http` in the output).
- Float comparison: `abs(a-b) <= abs_tol + rel_tol*abs(b)`; `NULL = NULL`.
- Columns with a declared tolerance are excluded from fingerprints and listed.
- Non-unique key on either side: FAIL with reason `key_not_unique`; schema and row-set steps still reported.

```python
"""Turn anything into a pyarrow.Table.

Strings are paths or URIs; objects that already look like tables pass through.
Every failure is a SourceError with the underlying message, so the CLI can exit 2.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any
from urllib.parse import parse_qs, urlparse

import duckdb
import pyarrow as pa


class SourceError(Exception):
    """A source could not be read."""


def load(source: Any) -> pa.Table:
    if isinstance(source, pa.Table):
        return source
    if hasattr(source, "to_arrow"):  # polars.DataFrame, bigquery RowIterator, ...
        return source.to_arrow()
    if hasattr(source, "to_pandas") is False and hasattr(source, "columns") and hasattr(source, "dtypes"):
        return pa.Table.from_pandas(source, preserve_index=False)
    if type(source).__name__ == "DataFrame":  # pandas
        return pa.Table.from_pandas(source, preserve_index=False)
    if isinstance(source, (str, Path)):
        return _load_str(str(source))
    raise SourceError(f"cannot read a source of type {type(source).__name__}")


def _load_str(s: str) -> pa.Table:
    if "://" in s:
        return _load_uri(s)
    p = Path(s)
    if not p.exists():
        raise SourceError(f"file not found: {s}")
    suffix = p.suffix.lower()
    con = duckdb.connect()
    try:
        if suffix == ".parquet":
            return con.execute("SELECT * FROM read_parquet(?)", [s]).arrow()
        if suffix in (".csv", ".tsv", ".txt"):
            return con.execute("SELECT * FROM read_csv_auto(?)", [s]).arrow()
    except duckdb.Error as e:
        raise SourceError(f"{s}: {e}") from e
    finally:
        con.close()
    raise SourceError(f"unsupported file type {suffix!r} (use .parquet or .csv, or a duckdb:// / sqlite:// / postgres:// URI)")


def _load_uri(uri: str) -> pa.Table:
    u = urlparse(uri)
    q = {k: v[0] for k, v in parse_qs(u.query).items()}
    table, sql = q.get("table"), q.get("sql")
    if not table and not sql:
        raise SourceError(f"{uri}: add ?table=NAME or ?sql=QUERY")
    query = sql or f'SELECT * FROM "{table}"'
    con = duckdb.connect()
    try:
        if u.scheme == "duckdb":
            path = u.path
            con.close()
            con = duckdb.connect(path, read_only=True)
            return con.execute(query).arrow()
        if u.scheme == "sqlite":
            con.execute("INSTALL sqlite; LOAD sqlite;")
            con.execute(f"ATTACH '{u.path}' AS src (TYPE sqlite, READ_ONLY)")
        elif u.scheme in ("postgres", "postgresql"):
            con.execute("INSTALL postgres; LOAD postgres;")
            dsn = uri.split("?")[0]
            con.execute(f"ATTACH '{dsn}' AS src (TYPE postgres, READ_ONLY)")
        else:
            raise SourceError(f"unsupported scheme {u.scheme!r}")
        con.execute("USE src")
        return con.execute(query).arrow()
    except duckdb.Error as e:
        raise SourceError(f"{uri}: {e}") from e
    finally:
        con.close()


def from_bigquery(sql: str, project: str | None = None) -> pa.Table:
    """Run a BigQuery query and return its result as Arrow. Needs cutover[bigquery]."""
    try:
        from google.cloud import bigquery
    except ImportError as e:  # pragma: no cover
        raise SourceError("pip install 'cutover[bigquery]' to use from_bigquery") from e
    client = bigquery.Client(project=project)
    return client.query(sql).result().to_arrow()
```

```python
"""The result of a comparison, and its text/JSON renderings."""

from __future__ import annotations

import json
from dataclasses import asdict, dataclass, field


@dataclass
class SchemaDiff:
    only_in_old: list[str]
    only_in_new: list[str]
    type_changes: list[tuple[str, str, str]]  # (column, old type, new type)
    compared: list[str]
    ignored: list[str]

    @property
    def empty(self) -> bool:
        return not (self.only_in_old or self.only_in_new or self.type_changes)


@dataclass
class KeyCheck:
    old_nulls: int
    new_nulls: int
    old_duplicates: int
    new_duplicates: int
    samples: list[dict]

    @property
    def unique(self) -> bool:
        return self.old_duplicates == 0 and self.new_duplicates == 0


@dataclass
class RowSet:
    only_in_old: int
    only_in_new: int
    matched: int
    samples_old: list[dict]
    samples_new: list[dict]


@dataclass
class ColumnDiff:
    column: str
    mismatches: int
    tolerance: str | None
    samples: list[dict]  # {key..., "old": x, "new": y}


@dataclass
class Partitions:
    column: str | None
    total: int
    identical: int
    differing: list[str]
    excluded_columns: list[str]


@dataclass
class Report:
    passed: bool
    reasons: list[str]
    key: list[str]
    old_rows: int
    new_rows: int
    schema: SchemaDiff
    keys: KeyCheck
    rows: RowSet
    columns: list[ColumnDiff]
    partitions: Partitions
    meta: dict = field(default_factory=dict)  # names of the sources, timestamp, version

    @property
    def clean_columns(self) -> int:
        return sum(1 for c in self.columns if c.mismatches == 0)

    def to_dict(self) -> dict:
        return asdict(self)

    def to_json(self, indent: int | None = 2) -> str:
        return json.dumps(self.to_dict(), indent=indent, default=str)

    def to_text(self) -> str:
        verdict = "PASS" if self.passed else "FAIL"
        lines = [
            f"cutover: {verdict}",
            f"rows: old {self.old_rows:,} · new {self.new_rows:,} · matched on key {self.rows.matched:,}"
            f" · only in old {self.rows.only_in_old:,} · only in new {self.rows.only_in_new:,}",
            f"columns: {len(self.columns) - self.clean_columns} / {len(self.columns)} columns with mismatches",
        ]
        for c in self.columns:
            if c.mismatches:
                lines.append(f"  {c.column}: {c.mismatches:,} mismatches" + (f" (tolerance {c.tolerance})" if c.tolerance else ""))
        if self.partitions.column:
            lines.append(
                f"fingerprints: {self.partitions.identical} / {self.partitions.total} partitions identical by {self.partitions.column}"
            )
            if self.partitions.differing:
                lines.append("  differing: " + ", ".join(self.partitions.differing[:20]))
        else:
            lines.append(f"fingerprint: {'identical' if self.partitions.identical == self.partitions.total else 'different'}")
        if not self.schema.empty:
            lines.append(
                f"schema: only in old {self.schema.only_in_old} · only in new {self.schema.only_in_new}"
                f" · type changes {self.schema.type_changes}"
            )
        if not self.keys.unique or self.keys.old_nulls or self.keys.new_nulls:
            lines.append(
                f"key: duplicates old {self.keys.old_duplicates} new {self.keys.new_duplicates}"
                f" · nulls old {self.keys.old_nulls} new {self.keys.new_nulls}"
            )
        if self.reasons:
            lines.append("reasons: " + ", ".join(self.reasons))
        return "\n".join(lines)

    def to_html(self) -> str:
        from cutover.html import render

        return render(self)
```

```python
import datetime as dt

import pyarrow as pa
import pytest


def base_table(n=40, days=4):
    """n rows over `days` days, key id, an int, a float, a string, a nullable, a timestamp."""
    ids = list(range(1, n + 1))
    day = [dt.date(2026, 1, 1) + dt.timedelta(days=i % days) for i in range(n)]
    return pa.table(
        {
            "id": ids,
            "day": day,
            "qty": [i % 7 for i in ids],
            "amount": [round(i * 1.25, 2) for i in ids],
            "name": [f"row-{i}" for i in ids],
            "note": [None if i % 5 == 0 else f"n{i}" for i in ids],
            "ts": [dt.datetime(2026, 1, 1, 12, 0) + dt.timedelta(minutes=i) for i in ids],
        }
    )


@pytest.fixture
def old():
    return base_table()


def with_rows_dropped(t: pa.Table, ids: list[int]) -> pa.Table:
    import pyarrow.compute as pc

    return t.filter(pc.invert(pc.is_in(t["id"], value_set=pa.array(ids))))


def with_column_replaced(t: pa.Table, name: str, values: list) -> pa.Table:
    i = t.schema.get_field_index(name)
    return t.set_column(i, name, pa.array(values, type=t.schema.field(name).type if name != "qty" else None))


def with_float_drift(t: pa.Table, name: str, delta: float, ids: list[int]) -> pa.Table:
    col = t[name].to_pylist()
    idl = t["id"].to_pylist()
    new = [v + delta if i in ids else v for v, i in zip(col, idl, strict=True)]
    return with_column_replaced(t, name, new)
```

```python
import pytest

from conftest import base_table, with_column_replaced, with_rows_dropped
from cutover.compare import compare


def test_identical_passes(old):
    r = compare(old, base_table(), key=["id"])
    assert r.passed and r.reasons == []
    assert r.rows.matched == 40 and r.rows.only_in_old == 0 and r.rows.only_in_new == 0
    assert all(c.mismatches == 0 for c in r.columns)

    assert r.rows.only_in_old == 2 and r.rows.only_in_new == 0
    assert "rows_only_in_old" in r.reasons
```

```python
"""compare(): the six steps, as SQL over two Arrow tables registered in DuckDB."""

from __future__ import annotations

import datetime as dt
from typing import Any

import duckdb
import pyarrow as pa

from cutover import __version__
from cutover.report import ColumnDiff, KeyCheck, Partitions, Report, RowSet, SchemaDiff
from cutover.sources import load

Tolerance = float | tuple[str, float]
FLOAT_TYPES = ("float", "double", "decimal")


def q(name: str) -> str:
    """Quote an identifier for DuckDB."""
    return '"' + name.replace('"', '""') + '"'


def compare(
    old: Any,
    new: Any,
    *,
    key: list[str],
    partition: str | None = None,
    tolerance: dict[str, Tolerance] | None = None,
    ignore: tuple[str, ...] | list[str] = (),
    sample: int = 20,
    allow_schema_drift: bool = False,
    memory_limit: str | None = None,
) -> Report:
    t_old, t_new = load(old), load(new)
    tolerance = dict(tolerance or {})
    ignore = list(ignore)
    for k in key:
        if k not in t_old.column_names or k not in t_new.column_names:
            raise ValueError(f"key column {k!r} missing from one side")
    if partition and (partition not in t_old.column_names or partition not in t_new.column_names):
        raise ValueError(f"partition column {partition!r} missing from one side")

    con = duckdb.connect()
    if memory_limit:
        con.execute(f"SET memory_limit='{memory_limit}'")
    con.register("old", t_old)
    con.register("new", t_new)
    reasons: list[str] = []

    # 1 · schema
    schema = _schema(t_old, t_new, ignore)
    if not schema.empty and not allow_schema_drift:
        reasons.append("schema_drift")

    # 2 · key
    keys = _keys(con, key, sample)
    if keys.old_nulls or keys.new_nulls:
        reasons.append("key_nulls")
    if not keys.unique:
        reasons.append("key_not_unique")

    # 3 · row set
    rows = _rows(con, key, sample)
    if rows.only_in_old:
        reasons.append("rows_only_in_old")
    if rows.only_in_new:
        reasons.append("rows_only_in_new")

    # 4 · columns (only meaningful on a unique key)
    value_cols = [c for c in schema.compared if c not in key and c != partition]
    columns: list[ColumnDiff] = []
    if keys.unique:
        columns = _columns(con, key, value_cols, tolerance, t_old.schema, sample)
        reasons += [f"column_mismatch:{c.column}" for c in columns if c.mismatches]

    # 5 · fingerprints
    partitions = _partitions(con, key, [c for c in schema.compared if c not in key], partition, tolerance)
    if partitions.identical != partitions.total:
        reasons.append("partition_fingerprint")

    con.close()
    return Report(
        passed=not reasons,
        reasons=reasons,
        key=list(key),
        old_rows=t_old.num_rows,
        new_rows=t_new.num_rows,
        schema=schema,
        keys=keys,
        rows=rows,
        columns=columns,
        partitions=partitions,
        meta={
            "old": getattr(old, "name", None) or (old if isinstance(old, str) else "old"),
            "new": getattr(new, "name", None) or (new if isinstance(new, str) else "new"),
            "partition": partition,
            "generated": dt.datetime.now(dt.timezone.utc).isoformat(timespec="seconds"),
            "cutover": __version__,
        },
    )


def _schema(t_old: pa.Table, t_new: pa.Table, ignore: list[str]) -> SchemaDiff:
    o = {f.name: str(f.type) for f in t_old.schema if f.name not in ignore}
    n = {f.name: str(f.type) for f in t_new.schema if f.name not in ignore}
    only_old = [c for c in o if c not in n]
    only_new = [c for c in n if c not in o]
    changes = [(c, o[c], n[c]) for c in o if c in n and o[c] != n[c]]
    compared = [c for c in o if c in n and o[c] == n[c]]
    ignored = [c for c in ignore if c in t_old.column_names or c in t_new.column_names]
    return SchemaDiff(only_old, only_new, changes, compared, ignored)


def _keys(con: duckdb.DuckDBPyConnection, key: list[str], sample: int) -> KeyCheck:
    ks = ", ".join(q(k) for k in key)
    null_pred = " OR ".join(f"{q(k)} IS NULL" for k in key)

    def nulls(t: str) -> int:
        return con.execute(f"SELECT count(*) FROM {t} WHERE {null_pred}").fetchone()[0]

    def dups(t: str) -> tuple[int, list[dict]]:
        rows = con.execute(
            f"SELECT {ks}, count(*) AS n FROM {t} GROUP BY {ks} HAVING count(*) > 1 ORDER BY n DESC LIMIT {sample}"
        ).fetchall()
        total = con.execute(
            f"SELECT coalesce(sum(n), 0) FROM (SELECT count(*) AS n FROM {t} GROUP BY {ks} HAVING count(*) > 1)"
        ).fetchone()[0]
        return int(total), [dict(zip([*key, "n"], r, strict=True)) for r in rows]

    od, os_ = dups("old")
    nd, ns = dups("new")
    return KeyCheck(nulls("old"), nulls("new"), od, nd, [{"side": "old", **s} for s in os_] + [{"side": "new", **s} for s in ns])


def _rows(con: duckdb.DuckDBPyConnection, key: list[str], sample: int) -> RowSet:
    ks = ", ".join(q(k) for k in key)

    def only(a: str, b: str) -> tuple[int, list[dict]]:
        cnt = con.execute(f"SELECT count(*) FROM (SELECT DISTINCT {ks} FROM {a} EXCEPT SELECT DISTINCT {ks} FROM {b})").fetchone()[0]
        smp = con.execute(
            f"SELECT {ks} FROM (SELECT DISTINCT {ks} FROM {a} EXCEPT SELECT DISTINCT {ks} FROM {b}) ORDER BY {ks} LIMIT {sample}"
        ).fetchall()
        return int(cnt), [dict(zip(key, r, strict=True)) for r in smp]

    oo, so = only("old", "new")
    on, sn = only("new", "old")
    matched = con.execute(
        f"SELECT count(*) FROM (SELECT DISTINCT {ks} FROM old INTERSECT SELECT DISTINCT {ks} FROM new)"
    ).fetchone()[0]
    return RowSet(oo, on, int(matched), so, sn)


def _columns(con, key, cols, tolerance, schema, sample) -> list[ColumnDiff]:
    return []  # Task 5


def _partitions(con, key, cols, partition, tolerance) -> Partitions:
    return Partitions(partition, 1, 1, [], [c for c in cols if c in tolerance])  # Task 5
```

```python
def _tol(t: Tolerance) -> tuple[str, float]:
    if isinstance(t, (int, float)):
        return "abs", float(t)
    kind, val = t
    if kind not in ("abs", "rel"):
        raise ValueError(f"tolerance kind must be 'abs' or 'rel', got {kind!r}")
    return kind, float(val)


def _is_float(schema: pa.Schema, col: str) -> bool:
    return any(str(schema.field(col).type).startswith(p) for p in FLOAT_TYPES)


def _columns(
    con: duckdb.DuckDBPyConnection,
    key: list[str],
    cols: list[str],
    tolerance: dict[str, Tolerance],
    schema: pa.Schema,
    sample: int,
) -> list[ColumnDiff]:
    for c in tolerance:
        if c not in cols:
            raise ValueError(f"tolerance given for {c!r}, which is not a compared value column")
        if not _is_float(schema, c):
            raise ValueError(f"tolerance given for {c!r}, which is not a float/decimal column")
    on = " AND ".join(f"o.{q(k)} IS NOT DISTINCT FROM n.{q(k)}" for k in key)
    ks = ", ".join(f"o.{q(k)} AS {q(k)}" for k in key)
    out: list[ColumnDiff] = []
    for c in cols:
        oc, nc = f"o.{q(c)}", f"n.{q(c)}"
        label = None
        if c in tolerance:
            kind, val = _tol(tolerance[c])
            label = f"{kind} {val:g}"
            bound = f"{val!r}" if kind == "abs" else f"{val!r} * abs({nc})"
            same = f"(({oc} IS NULL AND {nc} IS NULL) OR ({oc} IS NOT NULL AND {nc} IS NOT NULL AND abs({oc} - {nc}) <= {bound}))"
        else:
            same = f"({oc} IS NOT DISTINCT FROM {nc})"
        base = f"FROM old o JOIN new n ON {on} WHERE NOT {same}"
        count = con.execute(f"SELECT count(*) {base}").fetchone()[0]
        samples: list[dict] = []
        if count:
            rows = con.execute(
                f"SELECT {ks}, {oc} AS old, {nc} AS new {base} ORDER BY {', '.join(f'o.{q(k)}' for k in key)} LIMIT {sample}"
            ).fetchall()
            samples = [dict(zip([*key, "old", "new"], r, strict=True)) for r in rows]
        out.append(ColumnDiff(c, int(count), label, samples))
    return out


def _partitions(
    con: duckdb.DuckDBPyConnection,
    key: list[str],
    cols: list[str],
    partition: str | None,
    tolerance: dict[str, Tolerance],
) -> Partitions:
    excluded = [c for c in cols if c in tolerance]
    hashed = [c for c in cols if c not in excluded]
    row = " || '\x1f' || ".join(f"coalesce(CAST({q(c)} AS VARCHAR), '\x00')" for c in hashed) or "''"
    order = ", ".join(q(k) for k in key)
    part = f"CAST({q(partition)} AS VARCHAR)" if partition else "'all'"

    def fingerprints(t: str) -> dict[str, str]:
        rows = con.execute(
            f"SELECT p, md5(string_agg(r, '\n' ORDER BY {order})) FROM "
            f"(SELECT {part} AS p, {order}, {row} AS r FROM {t}) GROUP BY p"
        ).fetchall()
        return {str(p): h for p, h in rows}

    fo, fn = fingerprints("old"), fingerprints("new")
    parts = sorted(set(fo) | set(fn))
    differing = [p for p in parts if fo.get(p) != fn.get(p)]
    return Partitions(partition, len(parts), len(parts) - len(differing), differing, excluded)
```

```python
def _table(rows: list[dict], cols: list[str] | None = None, numeric: tuple[str, ...] = ()) -> str:
    if not rows:
        return '<p class="m">none</p>'
    cols = cols or list(rows[0].keys())
    head = "".join(f'<th class="{"n" if c in numeric else ""}">{escape(str(c))}</th>' for c in cols)
    body = "".join(
        "<tr>" + "".join(f'<td class="{"n" if c in numeric else ""}">{escape("∅" if r.get(c) is None else str(r.get(c)))}</td>' for c in cols) + "</tr>"
        for r in rows
    )
    return f'<div class="wrap"><table><thead><tr>{head}</tr></thead><tbody>{body}</tbody></table></div>'


def render(r: Report) -> str:
    cls = "ok" if r.passed else "bad"
    verdict = "PASS" if r.passed else "FAIL"
    meta = r.meta
    bad_cols = [c for c in r.columns if c.mismatches]
    parts = []
    parts.append(f"<h1>cutover · {escape(str(meta.get('old', 'old')))} → {escape(str(meta.get('new', 'new')))}</h1>")
    parts.append(
        f'<p class="m">key {escape(", ".join(r.key))}'
        + (f" · partition {escape(str(r.partitions.column))}" if r.partitions.column else "")
        + f" · {escape(str(meta.get('generated', '')))} · cutover {escape(str(meta.get('cutover', '')))}</p>"
    )
    parts.append(f'<p class="v {cls}">{verdict}</p>')
    if r.reasons:
        parts.append("<ul>" + "".join(f"<li><code>{escape(x)}</code></li>" for x in r.reasons) + "</ul>")
    parts.append("<h2>summary</h2>")
    parts.append(
        _table(
            [
                {"what": "rows old / new", "value": f"{r.old_rows:,} / {r.new_rows:,}"},
                {"what": "matched on key", "value": f"{r.rows.matched:,}"},
                {"what": "only in old / only in new", "value": f"{r.rows.only_in_old:,} / {r.rows.only_in_new:,}"},
                {"what": "columns with mismatches", "value": f"{len(bad_cols)} / {len(r.columns)}"},
                {
                    "what": "partitions identical" if r.partitions.column else "fingerprint",
                    "value": f"{r.partitions.identical} / {r.partitions.total}"
                    if r.partitions.column
                    else ("identical" if r.partitions.identical == r.partitions.total else "different"),
                },
            ],
            ["what", "value"],
        )
    )
    parts.append("<h2>1 · schema</h2>")
    s = r.schema
    parts.append(
        _table(
            [{"kind": "only in old", "columns": ", ".join(s.only_in_old) or "—"},
             {"kind": "only in new", "columns": ", ".join(s.only_in_new) or "—"},
             {"kind": "type changes", "columns": ", ".join(f"{c}: {a} → {b}" for c, a, b in s.type_changes) or "—"},
             {"kind": "ignored", "columns": ", ".join(s.ignored) or "—"},
             {"kind": "compared", "columns": ", ".join(s.compared)}],
            ["kind", "columns"],
        )
    )
    parts.append("<h2>2 · key</h2>")
    k = r.keys
    parts.append(
        _table(
            [{"side": "old", "nulls": k.old_nulls, "duplicate rows": k.old_duplicates},
             {"side": "new", "nulls": k.new_nulls, "duplicate rows": k.new_duplicates}],
            ["side", "nulls", "duplicate rows"], ("nulls", "duplicate rows"),
        )
    )
    if k.samples:
        parts.append(_table(k.samples))
    parts.append("<h2>3 · row set</h2>")
    parts.append(f'<p class="m">only in old: {r.rows.only_in_old:,}</p>' + _table(r.rows.samples_old))
    parts.append(f'<p class="m">only in new: {r.rows.only_in_new:,}</p>' + _table(r.rows.samples_new))
    parts.append("<h2>4 · columns</h2>")
    parts.append(
        _table(
            [{"column": c.column, "mismatches": f"{c.mismatches:,}", "tolerance": c.tolerance or "exact"} for c in r.columns],
            ["column", "mismatches", "tolerance"], ("mismatches",),
        )
    )
    for c in bad_cols:
        parts.append(f"<h3><code>{escape(c.column)}</code> · first {len(c.samples)} of {c.mismatches:,}</h3>" + _table(c.samples))
    parts.append("<h2>5 · fingerprints</h2>")
    p = r.partitions
    if p.column:
        parts.append(f'<p class="m">{p.identical} / {p.total} partitions identical by <code>{escape(p.column)}</code></p>')
        if p.differing:
            parts.append(_table([{"partition": d} for d in p.differing]))
    else:
        parts.append(f'<p class="m">{"identical" if p.identical == p.total else "different"}</p>')
    if p.excluded_columns:
        parts.append(f'<p class="m">excluded from the hash (tolerance declared): {escape(", ".join(p.excluded_columns))}</p>')
    return (
        "<!doctype html><html lang=\"en\"><head><meta charset=\"utf-8\">"
        '<meta name="viewport" content="width=device-width,initial-scale=1">'
        f"<title>cutover · {verdict}</title><style>{CSS}</style></head><body>" + "".join(parts) + "</body></html>"
    )
```

```python
"""cutover OLD NEW --key k [...] : exit 0 PASS, 1 FAIL, 2 error."""

from __future__ import annotations

import argparse
import sys
from pathlib import Path

from cutover import __version__
from cutover.compare import Tolerance, compare
from cutover.sources import SourceError


def parse_tol(s: str) -> tuple[str, Tolerance]:
    if "=" not in s:
        raise argparse.ArgumentTypeError("use --tol column=0.01 or --tol column=rel:1e-6")
    col, val = s.split("=", 1)
    if ":" in val:
        kind, num = val.split(":", 1)
        return col, (kind, float(num))
    return col, float(val)


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="cutover", description="Prove that nothing changed between two tables.")
    p.add_argument("old")
    p.add_argument("new")
    p.add_argument("--key", action="append", required=True, help="key column (repeat for a composite key)")
    p.add_argument("--partition", help="column to fingerprint by, e.g. a date")
    p.add_argument("--tol", action="append", type=parse_tol, default=[], help="column=0.01 (absolute) or column=rel:1e-6")
    p.add_argument("--ignore", action="append", default=[], help="column to leave out")
    p.add_argument("--sample", type=int, default=20)
    p.add_argument("--allow-schema-drift", action="store_true")
    p.add_argument("--memory", help="DuckDB memory limit, e.g. 4GB")
    p.add_argument("--json", type=Path, help="write the JSON report here")
    p.add_argument("--html", type=Path, help="write the HTML report here")
    p.add_argument("--quiet", action="store_true", help="print nothing; exit code only")
    p.add_argument("--version", action="version", version=f"cutover {__version__}")
    return p


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    try:
        report = compare(
            args.old,
            args.new,
            key=args.key,
            partition=args.partition,
            tolerance=dict(args.tol),
            ignore=args.ignore,
            sample=args.sample,
            allow_schema_drift=args.allow_schema_drift,
            memory_limit=args.memory,
        )
    except (SourceError, ValueError) as e:
        print(f"cutover: error: {e}", file=sys.stderr)
        return 2
    if args.json:
        args.json.write_text(report.to_json())
    if args.html:
        args.html.write_text(report.to_html())
    if not args.quiet:
        print(report.to_text())
    return 0 if report.passed else 1


if __name__ == "__main__":  # pragma: no cover
    sys.exit(main())
```

```python
"""Demo for texts/cutover.html: a 'legacy' pandas daily aggregate vs a 'new' SQL one over NYC yellow taxi, January 2025.

    .venv/bin/python tools/cutover_demo.py

Downloads one public parquet (~50 MB) into tools/data/, builds both aggregates, runs cutover,
writes assets/cutover/report.{html,json} and prints the text report. Whatever it finds is reported as found.
"""

from __future__ import annotations

import urllib.request
from pathlib import Path

import duckdb
import pandas as pd

import cutover

ROOT = Path(__file__).resolve().parent.parent
SRC = ROOT / "tools" / "data" / "yellow_tripdata_2025-01.parquet"
URL = "https://d37ci6vzurychx.cloudfront.net/trip-data/yellow_tripdata_2025-01.parquet"
OUT = ROOT / "assets" / "cutover"


def legacy(path: Path) -> pd.DataFrame:
    """The old way: pandas, local-time day, float sums, drop rows with no fare."""
    df = pd.read_parquet(path, columns=["tpep_pickup_datetime", "PULocationID", "fare_amount", "trip_distance"])
    df = df[df.fare_amount > 0]
    df["day"] = df.tpep_pickup_datetime.dt.date
    g = df.groupby(["day", "PULocationID"]).agg(trips=("fare_amount", "size"), fare=("fare_amount", "sum"), km=("trip_distance", "sum"))
    g["km"] = g["km"] * 1.609344
    return g.reset_index().rename(columns={"PULocationID": "zone"})


def new(path: Path) -> pd.DataFrame:
    """The new way: SQL in DuckDB. Same intent; independently written."""
    return duckdb.sql(
        f"""
        SELECT CAST(tpep_pickup_datetime AS DATE) AS day, PULocationID AS zone,
               count(*) AS trips, round(sum(fare_amount), 2) AS fare, sum(trip_distance) * 1.609344 AS km
        FROM read_parquet('{path}')
        WHERE fare_amount > 0
        GROUP BY 1, 2
        """
    ).df()


def main() -> None:
    SRC.parent.mkdir(parents=True, exist_ok=True)
    if not SRC.exists():
        print("downloading", URL)
        urllib.request.urlretrieve(URL, SRC)
    old, nw = legacy(SRC), new(SRC)
    old.name, nw.name = "legacy (pandas)", "new (SQL)"
    r = cutover.compare(old, nw, key=["day", "zone"], partition="day", tolerance={"km": 1e-6})
    OUT.mkdir(parents=True, exist_ok=True)
    (OUT / "report.html").write_text(r.to_html())
    (OUT / "report.json").write_text(r.to_json())
    print(r.to_text())


if __name__ == "__main__":
    main()
```
